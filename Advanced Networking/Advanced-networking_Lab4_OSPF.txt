--- MLS1 ---
en
conf t
hostname MLS1

vlan 10
name PC1
exit

vlan 20
name PC2
exit

ip routing

int Gi1/0/1
no switchport
ip address 192.168.60.6 255.255.255.252
description Network MLS1 with R1
exit

int Gi1/0/2
no switchport
ip address 192.168.60.1 255.255.255.252
description Network MLS1 with MLS2
exit

int vlan 10
ip address 192.168.10.1 255.255.255.0
standby 10 ip 192.168.10.3
standby 10 priority 150
standby 10 preempt
description Network vlan 10
exit

int vlan 20
ip address 192.168.20.1 255.255.255.0
standby 20 ip 192.168.20.3
standby 20 priority 100
standby 20 preempt
description Network vlan 20
exit

int range Gi1/0/3-4
switchport mode trunk
switchport trunk allowed vlan 10,20
exit

spanning-tree vlan 10 root primary
spanning-tree vlan 20 root secondary
