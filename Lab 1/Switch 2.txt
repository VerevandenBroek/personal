! VLANs

vlan 20

vlan 999

vlan 1000

exit

! VLAN interface + IP
interface vlan 10

exit

! Ongebruikte poorten → VLAN 999 (gesloten)
interface range fa0/3 - 17, fa0/19 - 24, g0/1 - 2

 switchport access vlan 999

exit

! Studentenpoort (VLAN 20)
interface fa0/18
 switchport mode access
 switchport access vlan 20
 no shutdown
exit
show vlan brief

! Trunks (EtherChannel op fa0/1-2)
interface range fa0/1 - 2
 switchport mode trunk
 switchport trunk native vlan 1000
 switchport trunk allowed vlan 10,20,1000
 channel-group 1 mode active
exit
show interfaces trunk
